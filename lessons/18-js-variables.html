المتغيّرات والشروط: كيف تمنح صفحتك قرارًا

اللغة التي تعرف أن تحفظ قيمة، وتقرّر بناءً عليها، هي لغة برمجة. في هذا الدرس تحديدًا يبدأ الفرق بينك وبين من ينسخ الأكواد.

الصندوق (المتغيّر) نحفظ فيه شيئًا: رقمًا أو كلمة. والشرط يقول: «إن كان الرقم أكبر من عشرة فاكتب كذا، وإلا فاكتب كذا». بهذين الشيئين وحدهما تستطيع أن تجعل صفحتك تفكّر.

ثلاثة أنواع تكفي في البداية

كل قيمة في JavaScript لها نوع. لا تحتاج إلى حفظ عشرة أنواع؛ ثلاثة تُغنيك الآن:

types.js
let age = 12;              // رقم (number) — بلا علامات تنصيص
let name = "سالم";          // نص (string) — بين علامتي تنصيص
let isReady = true;        // صحيح أو خطأ (boolean) — بلا تنصيص

console.log(typeof age);   // number
console.log(typeof name);  // string
console.log(typeof isReady); // boolean
أشهر خلط: الرقم بين تنصيصين

12 رقم، و"12" نص. الفرق ليس شكليًّا: "12" + 1 تعطي "121" (لصق نصوص)، و12 + 1 تعطي 13. إن ظهرت لك أرقام ملتصقة بلا معنى، فابحث عن تنصيص زائد.

إعلان المتغيّر وقواعده

  • الاسم بلا مسافات: userName صحيح، user name خطأ.
  • يبدأ بحرف أو _، لا برقم.
  • حسّاس لحالة الأحرف: Name ليست name.
  • اكتبه بالإنجليزية دائمًا — ولو كان مشروعك بالعربية. الأسماء العربية تُربك الأدوات.
  • اجعله معبّرًا: clicks أفضل من x، وtotalPrice أفضل من t.

المقارنة: كيف تسأل سؤالًا

compare.js
let age = 12;

console.log(age > 10);    // true
console.log(age === 12);  // true — يساوي؟ بثلاث علامات
console.log(age !== 15);  // true — لا يساوي
console.log(age < 5);     // false

// ونستطيع الجمع بين شرطين
console.log(age > 10 && age < 20);  // صحيح: الاثنان معًا
console.log(age > 20 || age < 15);  // صحيح: واحد منهما يكفي
===

«يساوي تمامًا»: نفس القيمة ونفس النوع. استخدم الثلاثة دائمًا.

==

«يساوي بعد التحويل»، فتعتبر "12" تساوي 12. تجنّبها: هي سبب أعطال صامتة كثيرة.

&&

«و» — الشرطان معًا. وإن فشل الأول، لا يُنظر إلى الثاني أصلًا.

||

«أو» — يكفي واحد منهما.

!

«عكس». !isReady تعني «ليس جاهزًا».

القرار: if و else

decide.js
let score = 7;

if (score >= 10) {
  console.log("ممتاز");
} else if (score >= 5) {
  console.log("مقبول");
} else {
  console.log("أعد المحاولة");
}

اقرأه كجملة إنجليزية-عربية: «إن كان score أكبر أو يساوي ١٠ فاطبع ممتاز، وإلا إن كان أكبر أو يساوي ٥ فاطبع مقبول، وإلا فاطبع أعد المحاولة». لاحظ أن المتصفح يقف عند أول شرط يتحقق ولا يكمل — ولهذا رتّب شروطك من الأصعب إلى الأسهل.

قرار حقيقي في الصفحة

الآن نربط كل هذا بالصفحة: نأخذ رقمًا من المستخدم، ونقرّر بناءً عليه.

grade.js
const input = document.getElementById("score");
const btn = document.getElementById("check");
const out = document.getElementById("result");

btn.addEventListener("click", function () {
  const value = Number(input.value);   // نحوّل النص إلى رقم

  if (input.value === "") {
    out.textContent = "اكتب رقمًا أولًا";
  } else if (value >= 50) {
    out.textContent = "ناجح";
  } else {
    out.textContent = "لم تنجح بعد — أعد المحاولة";
  }
});
لماذا Number(…)؟

كل ما يكتبه المستخدم في خانة هو نص، حتى لو كتب أرقامًا. فقبل أن تقارن، حوّله إلى رقم. وإن كتب حروفًا، فالنتيجة NaN أي «ليس رقمًا» — وسنعالج هذه الحالة في الدرس القادم.

أخطاء شائعة

  • ! استخدام = بدل === في الشرط: الأولى تُسنِد قيمة، والثانية تسأل.
  • ! نسيان الأقواس { }، فيُنفّذ سطر واحد فقط بدل الكتلة.
  • ! مقارنة نص برقم بلا Number().
  • ! تسمية المتغيّر باسم من كلمات محفوظة مثل let أو class.

تمرينك الآن

١٢ دقيقة · بلا مشاهدة حل

اصنع آلة عمر: خانة يكتب فيها المستخدم عمره، وزر، ونتيجة تقول: «طفل» إن كان أقل من ١٣، و«مراهق» من ١٣ إلى ١٩، و«بالغ» فوق ذلك. وإن كتب شيئًا ليس رقمًا، فاطبع «اكتب عمرًا بالأرقام».

أظهر الحل
age.js
const input = document.getElementById("age");
const btn = document.getElementById("go");
const out = document.getElementById("out");

btn.addEventListener("click", function () {
  const value = Number(input.value);

  if (input.value === "" || isNaN(value)) {
    out.textContent = "اكتب عمرًا بالأرقام";
  } else if (value < 13) {
    out.textContent = "طفل";
  } else if (value < 20) {
    out.textContent = "مراهق";
  } else {
    out.textContent = "بالغ";
  }
});

انتبه إلى ترتيب الشروط: لم نكتب «من ١٣ إلى ١٩» لأن الشرط الثاني لا يُفحص إلا إن فشل الأول. فحين نصل إليه نعرف أن العمر ١٣ أو أكثر. هذا تفكير يوفّر عليك كتابة وشروطًا متشابكة.

مراجعة سريعة

  1. ما الفرق بين 12 و"12"، ولماذا يهمّ؟
  2. لماذا نستخدم === لا ==؟
  3. لماذا نرتّب الشروط من الأصعب إلى الأسهل؟
  4. ما وظيفة Number() هنا؟
