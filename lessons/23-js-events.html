الأحداث: كيف تسمع الصفحة ما يفعله الزائر

حتى الآن صفحتك تتكلّم ولا تسمع. هذا الدرس يقلب ذلك: نقول للمتصفح «حين يضغط أحد هذا الزر، نفّذ هذه الدالة» — ثم نتعامل مع ما أرسله الحدث نفسه.

الحدث شيء يقع في الصفحة: ضغطة، كتابة حرف، انتهاء تحميل. والاستماع هو أن تسجّل دالة تقول «حين يقع هذا، نفّذني». المتصفح ينادي دالتك، ويعطيك معها تفاصيل ما وقع: مَن ضغط، وأيّ زر، وأين.

لماذا لا نكتب onclick في HTML

سترى في أمثلة قديمة <button onclick="doIt()">. تعمل، لكنها تخلط ثلاث لغات في سطر واحد: بنية الصفحة، وسلوكها، وأسلوبها. والنتيجة أنك تبحث عن الكود في مكانين، وتصعب عليك إضافة مستمع ثانٍ للزر نفسه.

bad.html
<!-- لا تفعل هذا: السلوك داخل البنية -->
<button onclick="alert('تم')">اضغط</button>

البديل سطر في JavaScript، والعنصر في HTML يبقى عنصرًا فقط:

good.js
const btn = document.querySelector("#save");

btn.addEventListener("click", function () {
  console.log("تمّ الحفظ");
});

ثلاثة أجزاء: العنصر، واسم الحدث، والدالة. والدالة هنا بلا اسم — تُسمّى «دالة مجهولة»، ولا تُنادى منك: المتصفح هو من يناديها حين يقع الحدث.

الحدث يحمل معه معلومات

المتصفح لا ينادي دالتك فارغة اليدين؛ يمرّر لها كائنًا اسمه event فيه ما وقع وأين:

event.js
document.querySelector("#save").addEventListener("click", function (e) {
  console.log(e.type);       // "click"
  console.log(e.target);     // العنصر الذي ضُغط فعلًا
  e.preventDefault();        // امنع السلوك الافتراضي إن وُجد
});
  • e.type — اسم الحدث. مفيد حين تخدم دالة واحدة عدة أحداث.
  • e.target — مصدر الحدث. هو مفتاح الحيلة في نهاية الدرس.
  • e.preventDefault() — يوقف ما يفعله المتصفح تلقائيًّا: إرسال نموذج، أو فتح رابط.
لماذا يختفي ما كتبته عند الضغط على إرسال؟

لأن النموذج يعيد تحميل الصفحة بسلوكه الافتراضي، فتضيع حالتك كلها. الحل ليس حذف الزر، بل e.preventDefault() في أوّل دالتك — ثم تُرسل أنت ما تريد متى أردت.

قراءة ما كتبه الزائر

في النماذج نقرأ value من الخانة. والحدث الأنسب هو submit على النموذج نفسه، لا click على الزر — لأن الزائر قد يُرسل بمفتاح Enter، وعندها لا ضغطة زر أصلًا.

form.js
const form = document.querySelector("#todo");
const input = document.querySelector("#task");
const list = document.querySelector("#list");

form.addEventListener("submit", function (e) {
  e.preventDefault();                 // أوقف إعادة التحميل

  const text = input.value.trim();    // اقرأ واقتطع الفراغات
  if (!text) return;                  // لا تُضف فراغًا

  const li = document.createElement("li");
  li.textContent = text;
  list.appendChild(li);

  input.value = "";                   // فرّغ الخانة للكتابة التالية
  input.focus();                      // أعد المؤشّر إليها
});

لاحظ السطر if (!text) return;. الحدّ الفاصل بين تمرين يعمل وأداة يستعملها أحد: امنع المدخل الفارغ قبل أن يصل إلى الشاشة.

أحداث ستستعملها كثيرًا

  • click — ضغطة أو نقرة. الأشهر.
  • submit — إرسال نموذج. على form لا على الزر.
  • input — كل تغيّر في خانة كتابة، حرفًا بحرف. للتصفية الفورية.
  • keydown — ضغط مفتاح. فيه e.key مثل "Escape".
  • change — تغيّر في قائمة اختيار أو مربّع اختيار.

حيلة واحدة تُغنيك عن عشرين مستمعًا

لو أضفت قائمة فيها ثلاثون عنصرًا، هل تضع مستمعًا لكل عنصر؟ لا. تضع مستمعًا واحدًا على الأب، ثم تسأل: من الجزء الذي وقع عليه الحدث؟

delegate.js
const list = document.querySelector("#list");

list.addEventListener("click", function (e) {
  if (e.target.tagName !== "LI") return;   // تجاهل ما ليس عنصر قائمة
  e.target.classList.toggle("done");       // بدّل حالة العنصر المضغوط
});

هذه الطريقة تسمّى «تفويض الحدث». ميزتها الحقيقية ليست الاختصار، بل أن العناصر التي ستُضاف لاحقًا تعمل بلا سطر إضافي — لأن المستمع على الأب لا على الأبناء.

أخطاء تتكرّر

  • المستمع قبل وجود العنصر: لو كان السكربت في head بلا defer، فالعنصر لم يُخلق بعد وquerySelector تُعيد null.
  • استدعاء الدالة بنفسك: addEventListener("click", save()) تُنفّذ save فورًا. المطلوب تمرير الاسم: save بلا قوسين.
  • نقل الكود داخل الدالة: الدالة تُنفّذ في كل ضغطة؛ ما لا يتغيّر يبقى خارجها.

تمرين: عدّاد بثلاثة أزرار

في المختبر: رقم يبدأ من صفر، وثلاثة أزرار: زيادة، إنقاص، تصفير. زر الزيادة يتوقّف عند ١٠ ويُخفي نفسه، وزر الإنقاص يتوقّف عند صفر. ثم أضف مفتاح Enter ليزيد أيضًا.

حلّ مقترح — افتحه بعد المحاولة
أظهر الحل
counter.html
<p id="out">٠</p>
<button id="plus">+</button>
<button id="minus">−</button>
<button id="reset">تصفير</button>
counter.js
const out = document.querySelector("#out");
const plus = document.querySelector("#plus");
const minus = document.querySelector("#minus");
let n = 0;

function paint() {
  out.textContent = n;
  plus.hidden = n >= 10;
  minus.disabled = n <= 0;
}

plus.addEventListener("click", function () { if (n < 10) n++; paint(); });
minus.addEventListener("click", function () { if (n > 0) n--; paint(); });
document.querySelector("#reset").addEventListener("click", function () { n = 0; paint(); });

document.addEventListener("keydown", function (e) {
  if (e.key === "Enter") { if (n < 10) n++; paint(); }
});

paint();

لاحظ paint() في النهاية: نرسم الحالة الحقيقية عند التحميل، لا نفترض أن الشاشة تساوي ما في المتغيّر.
