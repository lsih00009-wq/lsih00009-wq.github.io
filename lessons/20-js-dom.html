DOM والتخزين: صفحة تتذكّر

هذا الدرس ينهي المسار: صفحة تُنشئ عناصر بنفسها، وتحفظ ما فعله الزائر، فتجده عند عودته. بعده لن تكون قارئًا للويب، بل بانيًا له.

المتصفح يبني من صفحتك شجرة من الصناديق اسمها DOM. JavaScript تستطيع أن تمسك أي صندوق فيها، وتغيّر كلامه أو لونه، أو تُضيف صندوقًا جديدًا. والتخزين (localStorage) خزانة صغيرة في المتصفح تحفظ فيها قائمة أو كلمة، فتبقى موجودة بعد أن يُغلق الزائر الصفحة.

ما DOM بالضبط؟

حين يقرأ المتصفح ملف HTML، لا يبقى نصًّا: يبنيه شجرةً من العناصر، بعضها داخل بعض. body أب، وداخله header وmain، وداخل main فقرات وأزرار. هذه الشجرة هي DOM.

لماذا يهمّك؟ لأن JavaScript لا ترى ملفك، بل ترى هذه الشجرة الحيّة. حين تُغيّر عنصرًا فيها، يتغيّر ما يراه الزائر فورًا — بلا إعادة تحميل.

كيف تُمسك عنصرًا

select.js
// بمعرّف
const title = document.getElementById("title");

// بأي محدِّد CSS — وهذا الأقوى والأنسب
const firstBtn = document.querySelector(".btn");
const allBtns = document.querySelectorAll(".btn");

console.log(allBtns.length);   // عدد الأزرار
لماذا querySelector أفضل؟

لأنه يفهم نفس المحدِّدات التي تعلّمتها في CSS: querySelector(".note")، querySelector("#title")، querySelector("nav a"). قاعدة واحدة تخدم اللغتين، فلا تحفظ مرتين. وquerySelectorAll تُعيد قائمة كل العناصر المطابقة.

كيف تُغيّر ما فيه

change.js
const title = document.querySelector("#title");

title.textContent = "كلام جديد";      // غيّر النص
title.classList.add("active");        // أضف صنفًا من CSS
title.classList.toggle("hidden");     // أضف/احذف بالتناوب
title.style.color = "green";          // تغيير مباشر (نفضّله في CSS)

const input = document.querySelector("#name");
console.log(input.value);             // ما كتبه المستخدم
  • textContent — الكلام. آمن وواضح، واستخدمه دائمًا.
  • classList — الطريقة الصحيحة لتغيير الشكل: تُضيف صنفًا وتترك CSS تتصرّف.
  • style — تعمل، لكنها تخلط الشكل بالكود. لا تستخدمها إلا لحالة واحدة عابرة.
  • value — لما يكتبه المستخدم في خانة.

كيف تُنشئ عنصرًا جديدًا

هذه أهم خطوة: الصفحة التي تبني قائمتها بنفسها. الترتيب ثلاثة: اصنع، املأ، أضف.

create.js
const list = document.querySelector("#list");

function addItem(text) {
  const li = document.createElement("li");  // ١) اصنع
  li.textContent = text;                    // ٢) املأ
  list.appendChild(li);                     // ٣) أضف إلى الأب
}

addItem("خبز");
addItem("لبن");

لو نسيت الخطوة الثالثة، سيكون العنصر موجودًا فعلًا لكن لا يراه أحد — لأنه ليس داخل الشجرة. هذا الخطأ يحدث كثيرًا: الكود يعمل بلا رسالة خطأ، والشاشة فارغة.

التخزين: كيف تتذكّر الصفحة

localStorage خزانة داخل المتصفح: تحفظ فيها نصًّا فقط، ويبقى بعد الإغلاق. ولأنها تحفظ نصوصًا فقط، نحوّل القائمة إلى نص عند الحفظ، ونحوّلها مرة أخرى عند القراءة:

store.js
// حفظ
const items = ["خبز", "لبن"];
localStorage.setItem("items", JSON.stringify(items));

// قراءة
const saved = localStorage.getItem("items");
const list = saved ? JSON.parse(saved) : [];
console.log(list);   // ["خبز", "لبن"]
  • JSON.stringify — يحوّل القائمة إلى نص ليُحفظ.
  • JSON.parse — يعيد النص إلى قائمة.
  • الشرط saved ? … : [] يعني: إن كانت هناك بيانات سابقة فاقرأها، وإلا فابدأ بقائمة فارغة. بلا هذا الشرط سيُعطيك null في أول زيارة.
حدود التخزين

localStorage تخصّ متصفحًا واحدًا وجهازًا واحدًا: من يفتح موقعك من هاتف آخر لن يجد بياناته. وسعتها محدودة (بضعة ميجابايت)، ومن يمسح بيانات المتصفح يمحيها. هي للمفضّلات والمسودّات، لا للبيانات المهمة.

المشروع: قائمة مهام صغيرة

كل ما سبق في ملف واحد. اقرأه من أعلاه إلى أسفله، وستجد كل سطر مألوفًا:

app.js
const input = document.querySelector("#task");
const btn = document.querySelector("#add");
const list = document.querySelector("#list");

// ١) اقرأ ما حُفظ سابقًا، أو ابدأ فارغًا
let tasks = JSON.parse(localStorage.getItem("tasks") || "[]");

// ٢) ارسم القائمة على الشاشة
function render() {
  list.innerHTML = "";                    // نظّف أولًا
  tasks.forEach(function (task, index) {
    const li = document.createElement("li");
    li.textContent = task;

    li.addEventListener("click", function () {
      tasks.splice(index, 1);             // احذف هذه المهمة
      localStorage.setItem("tasks", JSON.stringify(tasks));
      render();
    });

    list.appendChild(li);
  });
}

// ٣) أضف مهمة جديدة
btn.addEventListener("click", function () {
  const text = input.value.trim();
  if (text === "") return;                // لا تُضف فراغًا

  tasks.push(text);
  input.value = "";                       // فرّغ الخانة
  localStorage.setItem("tasks", JSON.stringify(tasks));
  render();
});

render();   // ارسم عند فتح الصفحة
innerHTML = ""

حذف كل الأولاد. نستخدمها لتبدأ الرسم من جديد بلا تكرار.

trim()

تحذف المسافات من الطرفين، فتمنع أن تكون المهمة مسافة فارغة.

splice(index, 1)

تحذف عنصرًا واحدًا من القائمة عند موضعه.

return;

تخرج من الدالة فورًا. هنا: لا تُكمل إن كانت الخانة فارغة.

أخطاء شائعة

  • ! إنشاء العنصر ونسيان appendChild، فلا يظهر شيء.
  • ! نسيان JSON.parse عند القراءة، فيصبح التخزين نصًّا غير قابل للاستخدام.
  • ! الحفظ داخل الحلقة بدل الحفظ مرة واحدة بعد التغيير.
  • ! استخدام index داخل حدث بعد تغيّر القائمة، فيُحذف العنصر الخطأ.
  • ! وضع السكربت قبل العناصر، فيكون querySelector بلا نتيجة (null).

تمرينك الآن

٢٠ دقيقة · المشروع نفسه

ابنِ قائمة المهام كما في المثال، ثم أضف: زرًّا يمسح كل المهام، وعدّادًا يكتب «لديك ٣ مهام». ثم أغلق الصفحة وافتحها — يجب أن تجد مهامك كما تركتها.

أظهر الحل
extra.js
const counter = document.querySelector("#count");
const clearBtn = document.querySelector("#clear");

function render() {
  list.innerHTML = "";
  tasks.forEach(function (task, index) {
    const li = document.createElement("li");
    li.textContent = task;
    li.addEventListener("click", function () {
      tasks.splice(index, 1);
      localStorage.setItem("tasks", JSON.stringify(tasks));
      render();
    });
    list.appendChild(li);
  });

  counter.textContent = "لديك " + tasks.length + " مهام";
}

clearBtn.addEventListener("click", function () {
  tasks = [];
  localStorage.setItem("tasks", JSON.stringify(tasks));
  render();
});

لاحظ أن كل التغييرات تمرّ عبر مكانين فقط: القائمة tasks ثم render(). هذه العادة — «بيانات ثم رسم» — هي التي تفصل الكود الذي يعيش عن الكود الذي ينهار بعد أسبوع.
